def reason(u):
    identity = u['id']
    f = files[u['path']]
    src = read(R / f['source'])
    label = truth[identity]
    if identity in observations:
        return observations[identity]
    if u['ruleId'] == 'DEP_PQC_CAPABLE_LIBRARY':
        text = src.decode('utf8')
        if text.startswith(('{ =', '{broken')) or text.rstrip().endswith('['):
            return 'The source is an incomplete or invalid dependency/SBOM document and does not establish a library record. Its original unsupported label remains distinct from a safe negative.'
        if 'cryptographic-asset' in text:
            return 'The CycloneDX 1.7 component is a cryptographic-asset named ML-KEM-768, not a library/package dependency record in the bounded ingestion contract. The original unsupported dependency question is retained.'
        names = [('org.bouncycastle', 'BouncyCastle'), ('github.com/cloudflare/circl', 'CIRCL'), ('cryptography', 'cryptography'), ('openssl', 'openssl'), ('jose', 'jose'), ('ordinary-package', 'ordinary-package'), ('<groupId>ordinary</groupId>', 'ordinary-package')]
        name = next(display for token, display in names if token in text)
        if name == 'jose':
            return 'The source explicitly records jose 6.0.0, but this is JOSE dependency presence and does not establish the nominated standard-PQC-capable-library premise. The original ambiguity is retained; documentation shortening does not resolve it.'
        versions = ', '.join(label['parameters'])
        if name == 'ordinary-package':
            return f'The document records ordinary-package with version/constraint {versions}. This unrelated package does not establish the nominated capable-library property; installation and safety are not inferred.'
        return f'The document explicitly records {name} with version/constraint {versions}. This remains a supported static dependency-presence question without enabled-algorithm or installed-state inference.'
    if u['ruleId'].startswith('PRIVATE_KEY_'):
        text = src.decode('ascii')
        if text.startswith('-----BEGIN '):
            return 'A complete recognized private-key BEGIN header is present with a deliberately invalid synthetic body. The retained property concerns the header only and does not validate or decode a key.'
        if text.startswith('---BEGIN '):
            return 'The incomplete BEGIN-like line lacks the complete recognized private-key delimiter. This is a retained negative for the marker property, not absence-of-secret proof.'
        return 'The source contains only public-document text and no complete recognized private-key header. The negative remains limited to that exact marker property.'
    if f['artifact'] == 'certificate':
        s = structures[f['path']]
        if u['ruleId'] == 'CERT_P12_CONTAINER':
            if s.get('parsedAttributes') is False:
                return 'The .p12/.pfx suffix retains the frozen legacy container signal even though the bytes have no readable PFX structure. This does not establish valid container or inner certificate attributes.'
            return 'The source has a definite-length PFX v3 Data wrapper. The retained container-presence signal is separate from inner attributes, decryption, MAC integrity and trust.'
        if identity in excluded:
            return ('The definite-length PFX v3 outer Data contains AuthenticatedSafe content OID 1.2.840.113549.1.7.6, encryptedData, with no readable public certBag. '
                    'The nominated inner signature/SPKI attribute requires decryption, which the current EN/JA coverage, STATIC-ANALYSIS PKCS#12 subset and non-decrypting pkcs12.rs contract exclude. '
                    'Historical conversion provenance identifies a source certificate but does not make its encrypted inner attributes readable from this input.')
        if s.get('parsedAttributes') is False:
            return 'The bytes have no readable certificate/PFX structure and yield no public signature attribute. The original unsupported attribute question remains retained, without a safety inference.'
        if 'readablePublicCertificates' in s:
            assert len(s['readablePublicCertificates']) == 1
            s = s['readablePublicCertificates'][0]
            prefix = 'The PFX contains Data and one X.509 certBag, OID 1.2.840.113549.1.12.10.1.3, with readable public certificate attributes. '
        else:
            prefix = 'The public certificate is readable as PEM/DER. '
        if u['ruleId'] == 'CERT_RSA_3072':
            return prefix + (f"Its SPKI OID is {s['publicKeyOid']} and the RSA modulus bit length is {s['rsaModulusBits']}, so the exact size question remains in the public subset." if 'rsaModulusBits' in s else f"Its SPKI OID is {s['publicKeyOid']}, not RSA; the original size-property negative remains supported.")
        return prefix + f"Its signature OID is {s['signatureOid']}. Signature and SPKI are separate facts, and the original exact-signature question remains supported without trust verification."
    if identity in {'x131','x132','x133','x134','x135','x136','x137','x138','x139'}:
        return ('The imported crypto generateKeyPair call takes asymmetricKeyType from the finite local list of ' + ', '.join(label['parameters']) + '. '
                'Capability guards and assertion branches do not erase the nominal reference; loop members are not multiplied into new independent questions.')
    if u['ruleId'] == 'PQC_WEBCRYPTO_API':
        if label['label'] == 'unsupported':
            return 'TSX is a supported grammar, but the generateKey name field stops before a value. The original unsupported algorithm question remains visible rather than a safe negative.'
        value = 'ML-KEM-768' if 'ML-KEM-768' in src.decode() else 'AES-GCM'
        return f'The TSX source supplies the explicit WebCrypto name object {value}. Its original nominal algorithm question remains supported; the boolean declaration does not change that reference.'
    if u['ruleId'] in {'PQC_NODE_API', 'PQC_JAVA_API'} and src.lstrip().startswith(b'//'):
        return 'The entire apparent factory/getInstance call is inside a line comment, including the visible algorithm literal. Its original comment-control negative remains in the source contract.'
    if u['ruleId'] == 'SSH_DSS_SIGALG' and src.lstrip().startswith(b'#'):
        return 'The HostKeyAlgorithms ssh-dss directive is hash-commented. It remains a disabled-setting negative, not an effective DSA selection.'
    if f['artifact'] == 'configuration':
        text = src.decode().strip()
        if text.endswith('"'):
            return 'The configuration fragment stops at an opening quote and has no decidable operand. Its original unsupported question remains distinct from a safe negative.'
        operand = ', '.join(label['parameters'])
        if u['ruleId'] in {'NGINX_SSL_CIPHERS_RSA','K8S_INGRESS_RSA_CIPHERS'}:
            return f'The cipher directive explicitly supplies {operand}. The question concerns a named RSA operand; an AESGCM alternative does not prove the expanded effective suite is quantum safe.'
        return f'The protocol directive explicitly supplies {operand}. The legacy-version selection question remains a static file-setting question; no deployed configuration or negotiated connection is inferred.'
    raise RuntimeError('Missing source-authored reason for ' + identity)


def write_json(name, value):
    (O / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


unit_rows = []
for u in c['units']:
    original = truth[u['id']]
    f = files[u['path']]
    row = dict(u, source=f['source'], sourceSha256=f['sha256'],
               nominalLabel=original['label'], nominalAnnotations={k: original[k] for k in ['algorithms','usage','artifact','caseKind','parameters']},
               originalLabelSourceLabels={n: values[u['id']]['label'] for n, values in label_sources.items()},
               applicability='unsupported' if u['id'] in excluded else 'retained',
               sourceReason=reason(u), reviewed=True)
    unit_rows.append(row)
assert len(unit_rows) == 246 and len({u['id'] for u in unit_rows}) == 246
assert {u['id'] for u in unit_rows if u['applicability'] == 'unsupported'} == excluded
file_rows = []
for f in c['files']:
    units = [u for u in unit_rows if u['path'] == f['path']]
    assert units
    row = dict(path=f['path'], source=f['source'], sha256=f['sha256'], bytes=len(read(R / f['source'])),
               inputFormat=f['inputFormat'], artifact=f['artifact'], reviewed=True,
               questionIds=[u['id'] for u in units],
               sourceReasons=list(dict.fromkeys(u['sourceReason'] for u in units)))
    if f['path'] in structures:
        row['publicStructure'] = structures[f['path']]
    file_rows.append(row)
assert len(file_rows) == 203
for name, rows in [('unit-audit.jsonl',unit_rows), ('file-audit.jsonl',file_rows)]:
    (O / name).write_text(''.join(json.dumps(v,ensure_ascii=False) + '\n' for v in rows))
