def sha(data):
    return hashlib.sha256(data).hexdigest()


def read(path):
    assert path.resolve() == path and path.is_file() and not path.is_symlink()
    assert path.stat().st_size <= 2 * 1024 * 1024
    return path.read_bytes()


def nodes(data):
    result = []
    offset = 0
    while offset < len(data):
        start = offset
        tag = data[offset]
        offset += 1
        assert tag & 31 != 31
        length = data[offset]
        offset += 1
        if length & 128:
            count = length & 127
            assert 0 < count <= 4 and offset + count <= len(data)
            length = int.from_bytes(data[offset:offset + count], 'big')
            offset += count
        end = offset + length
        assert end <= len(data)
        result.append((tag, data[offset:end], start, end))
        offset = end
    assert offset == len(data)
    return result


def one(data, tag):
    values = nodes(data)
    assert len(values) == 1 and values[0][0] == tag
    return values[0][1]


def oid(data):
    numbers = [data[0] // 40, data[0] % 40]
    value = 0
    for byte in data[1:]:
        value = value * 128 + (byte & 127)
        if not byte & 128:
            numbers.append(value)
            value = 0
    assert not value
    return '.'.join(str(v) for v in numbers)


def certificate(data):
    fields = nodes(one(data, 0x30))
    assert len(fields) == 3
    signature = oid(nodes(fields[1][1])[0][1])
    tbs = nodes(fields[0][1])
    pos = 1 if tbs[0][0] == 0xa0 else 0
    spki = nodes(tbs[pos + 5][1])
    alg = nodes(spki[0][1])
    algorithm = oid(alg[0][1])
    result = dict(signatureOid=signature, publicKeyOid=algorithm)
    if algorithm == '1.2.840.113549.1.1.1':
        assert spki[1][0] == 3 and spki[1][1][0] == 0
        key = nodes(one(spki[1][1][1:], 0x30))
        assert key[0][0] == 2
        result['rsaModulusBits'] = int.from_bytes(key[0][1], 'big').bit_length()
    elif len(alg) > 1 and alg[1][0] == 6:
        result['curveOid'] = oid(alg[1][1])
    return result


def content(data):
    values = nodes(data)
    assert values[0][0] == 6
    kind = oid(values[0][1])
    return kind, one(values[1][1], 4) if kind == '1.2.840.113549.1.7.1' else None


def pfx(data):
    fields = nodes(one(data, 0x30))
    assert fields[0][0] == 2 and int.from_bytes(fields[0][1], 'big') == 3
    outer_oid, auth = content(fields[1][1])
    assert outer_oid == '1.2.840.113549.1.7.1'
    entries = []
    bag_oids = []
    certs = []
    for entry in nodes(one(auth, 0x30)):
        assert entry[0] == 0x30
        kind, safe = content(entry[1])
        entries.append(kind)
        if safe is None:
            continue
        for bag in nodes(one(safe, 0x30)):
            assert bag[0] == 0x30
            values = nodes(bag[1])
            assert values[0][0] == 6 and values[1][0] == 0xa0
            bag_oid = oid(values[0][1])
            bag_oids.append(bag_oid)
            if bag_oid == '1.2.840.113549.1.12.10.1.3':
                certbag = nodes(one(values[1][1], 0x30))
                assert oid(certbag[0][1]) == '1.2.840.113549.1.9.22.1'
                certs.append(certificate(one(certbag[1][1], 4)))
    return dict(pfxVersion=3, outerContentOid=outer_oid, authenticatedSafeContentOids=entries,
                readableBagOids=bag_oids, readablePublicCertificates=certs,
                encryptedContentsOpaque=any(v == '1.2.840.113549.1.7.6' for v in entries),
                macPresent=len(fields) > 2, macVerified=False)
