def safe_line(text,line):
    lines=text.splitlines()
    if not 1<=line<=len(lines): return '[No textual line at original nominated position]'
    s=lines[line-1]
    if s=='SYNTHETIC_NOT_A_KEY': return '[Invalid marker body omitted]'
    return s
