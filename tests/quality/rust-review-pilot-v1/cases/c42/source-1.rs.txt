fn make_finding_id(
    rule_id: &str,
    file: &str,
    line: usize,
    column: usize,
    evidence: &str,
) -> String {
    crate::identity::native_id(
        "finding",
        &[
            rule_id,
            file,
            &line.to_string(),
            &column.to_string(),
            evidence,
        ],
    )
}
