+# Confined protocol verification
+
+Prepare tools and the exact locked Cargo cache separately:
+
+```sh
+python3 tests/protocol/prepare-image.py --out-dir target/protocol-image
+PQC_RUN_CONTAINER_TESTS=1 PQC_PROTOCOL_IMAGE_FILE=target/protocol-image/IMAGE.json \
+  python3 -m unittest discover -s tests/protocol -p 'test_*.py' -v
+python3 tests/protocol/run-isolated.py --out-dir target/protocol-ci \
+  --image-file target/protocol-image/IMAGE.json --previews --ssh-kex
+```
+
+Preparation may access official package/tool sources. It copies only manifests
+and empty stubs into its new image layer; Cargo fetch does not execute repository
+build scripts/tests. OpenSSH 10.2p1 is checksum-pinned. A cached tool image can be
+selected by immutable image ID using `prepare-image.py --tool-image sha256:...`.
+No target code runs until the offline container starts.
+
+Target compilation, all-feature tests/clippy, eight synthetic TLS/SSH report
+previews and five product OpenSSH cases share the same nonroot, read-only source
+and root, empty environment, no host HOME/agent/socket and loopback-only namespace.
+Container cgroup v2 bounds all descendants to 6 GiB memory (no added swap), two CPU
+quota and 128 PIDs. Incremental caches and debug symbols are disabled, while
+test/debug assertions remain enabled. All build/temp/HOME/evidence data lives in a 4 GiB tmpfs plus
+1 MiB shared memory. The host deadline is 1200 seconds. Every exit path removes
+the container. The host exports at most 64 MiB of regular, relative artifacts.
+Read HOST-RESULTS.json for actual kernel checks, immutable image and source hashes;
+RESULTS.json binds exact executables and their logs. Passing logs do not replace
+the separate resource-pressure and cleanup checks.
+
+Set PQC_PROTOCOL_EVIDENCE_DIR to preserve individual resource-check evidence.
+The tests trigger memory OOM across children, aggregate ENOSPC across files, PID
+exhaustion and a timeout, then check removal. Missing Docker, cache, image, cgroup
+limits or a required fixture fails the gate; there is no weaker fallback.
+
+`run-native-smoke.py` and `isolation.py` retain macOS development smoke checks.
+They do not enforce group memory or aggregate disk and are not the CI confinement
+gate. Previous source-bound records remain historical; current strict-run evidence
+belongs to its own output directory. This workflow does not publish or sign builds.

+def sha(path):
+    h = hashlib.sha256()
+    with Path(path).open('rb') as stream:
+        for part in iter(lambda: stream.read(65536), b''):
+            h.update(part)
+    return h.hexdigest()

+def cgroup(name):
+    value = (Path('/sys/fs/cgroup')/name).read_text().strip()
+    if value == 'max':
+        raise RuntimeError('finite cgroup v2 limits are required')
+    return int(value)

+def write(name, record):
+    (EVIDENCE/name).write_text(json.dumps(record,indent=2)+'\n')

+def facts():
+    if sys.platform != 'linux' or os.getuid() != 65534 or not ROOT.is_dir():
+        raise RuntimeError('requires the confined nonroot Linux launcher')
+    memory = cgroup('memory.max'); swap = cgroup('memory.swap.max'); pids = cgroup('pids.max')
+    cpu = (Path('/sys/fs/cgroup')/'cpu.max').read_text().split()

+        raise RuntimeError('CPU quota and zero additional swap are required')
+    active = {p.name for p in Path('/sys/class/net').iterdir()
+              if (p/'flags').is_file() and int((p/'flags').read_text(),16) & 1}
+    if active != {'lo'}:
+        raise RuntimeError('only loopback may be active in the isolated network namespace')
+    return {'memoryMax':memory, 'swapMax':swap, 'pidsMax':pids, 'cpuMax':cpu,
+            'scratchCapacity':os.statvfs(SCRATCH).f_blocks*os.statvfs(SCRATCH).f_frsize}

+def boundary():

+def digest(path):
+    h = hashlib.sha256()
+    with Path(path).open('rb') as stream:
+        for part in iter(lambda: stream.read(65536), b''):
+            h.update(part)
+    return h.hexdigest()

+def source_bytes(root, name, limit=32*1024**2):
+    """Resolve each repository component through held, no-follow directory FDs."""
+    path = PurePosixPath(name); parts = path.parts
+    if not parts or path.is_absolute() or '..' in parts or '\\' in name or ':' in name or any(ord(c)<32 for c in name):
+        raise ValueError('unsafe source path')
+    directory = os.open(root, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
+    try:
+        for part in parts[:-1]:
+            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory)
+            os.close(directory); directory = child
+        fd = os.open(parts[-1], os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=directory)
+        with os.fdopen(fd, 'rb') as stream:
+            info = os.fstat(stream.fileno())
+            if not stat.S_ISREG(info.st_mode) or info.st_size > limit:
+                raise ValueError('source must be a bounded regular file')
+            data = stream.read(limit+1)
+            if len(data) > limit:
+                raise ValueError('source grew beyond the size limit')
+            return data
+    finally:
+        os.close(directory)
+
+
+def snapshot(root, destination):
+    root = Path(root).resolve(); destination = Path(destination)
+    destination.mkdir(parents=True, exist_ok=False)
+    names = subprocess.check_output(['git', 'ls-files', '-z', '--cached', '--others', '--exclude-standard'], cwd=root).decode().split('\0')
+    bindings = {}; size = 0
+    for name in sorted(set(filter(None, names))):
+        path = PurePosixPath(name)
+        if path.is_absolute() or '..' in path.parts or any(ord(c) < 32 for c in name) or '\\' in name or ':' in name:
+            raise ValueError('unsafe source path')
+        if name not in ALLOWED_FILES and path.parts[0] not in ALLOWED_DIRS:
+            continue
+        if any(p == '.git' or p.startswith('.env') or p == '__pycache__' for p in path.parts):
+            continue
+        try:
+            data = source_bytes(root, name)
+        except OSError as error:
+            raise ValueError('source changed or contains a link/special entry') from error
+        size += len(data)
+        if size > 128*1024**2:
+            raise ValueError('source snapshot exceeds total size limit')
+        target = destination / name; target.parent.mkdir(parents=True, exist_ok=True)
+        target.write_bytes(data); target.chmod(0o444)
+        bindings[name] = hashlib.sha256(data).hexdigest()
+        if bindings[name] != hashlib.sha256(source_bytes(root,name)).hexdigest():
+            raise ValueError('source changed while copying')
+    return bindings

+def image_id(path):
+    value = json.loads(Path(path).read_text())['image']
+    if not re.fullmatch(r'sha256:[0-9a-f]{64}', value):
+        raise ValueError('a locally prepared immutable image ID is required')

+def extract_artifacts(data, destination, max_bytes=64*1024**2):
+    destination = Path(destination).resolve(); destination.mkdir(parents=True, exist_ok=True)
+    total = 0; names = set()
+    with tarfile.open(fileobj=io.BytesIO(data)) as archive:
+        members = []
+        for member in archive:
+            if len(members) >= 4096:
+                raise ValueError('too many artifact entries')
+            members.append(member)
+        for member in members:
+            name = PurePosixPath(member.name)
+            if name.is_absolute() or '..' in name.parts or any(ord(c) < 32 for c in member.name) or '\\' in member.name or ':' in member.name:
+                raise ValueError('unsafe artifact path')
+            if member.isdir():
+                continue
+            if member.type not in {tarfile.REGTYPE,tarfile.AREGTYPE} or not name.parts or member.size < 0 or name in names:
+                raise ValueError('artifact must be a unique regular file')
+            names.add(name); total += member.size
+            if total > max_bytes:
+                raise ValueError('artifact size limit exceeded')
+        for member in members:
+            if not member.isfile():
+                continue
+            target = destination / member.name
+            if any(parent.is_symlink() for parent in [target, *target.parents]):
+                raise ValueError('artifact destination cannot contain links')
+            target.parent.mkdir(parents=True, exist_ok=True)
+            with target.open('xb') as stream, archive.extractfile(member) as source:
+                shutil.copyfileobj(source, stream, 65536)

+def bounded_output(command, limit=64*1024**2, timeout=30):
+    process = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL)
+    chunks = []; size = 0; deadline = time.monotonic() + timeout
+    try:
+        with selectors.DefaultSelector() as selector:
+            selector.register(process.stdout, selectors.EVENT_READ)
+            while True:
+                remaining = deadline - time.monotonic()
+                if remaining <= 0:
+                    raise subprocess.TimeoutExpired(command, timeout)
+                if not selector.select(remaining):
+                    raise subprocess.TimeoutExpired(command, timeout)
+                part = os.read(process.stdout.fileno(), min(65536, limit-size+1))
+                if not part:
+                    break
+                chunks.append(part); size += len(part)
+                if size > limit:
+                    raise ValueError('container artifact export exceeds limit')
+        if process.wait(timeout=max(.1, deadline-time.monotonic())):
+            raise RuntimeError('container artifact export failed')
+        return b''.join(chunks)
+    finally:
+        if process.poll() is None:
+            process.kill()
+        process.wait(); process.stdout.close()
+

+        '(allow network-outbound (remote ip "localhost:*"))',

+    for key, cap in [(resource.RLIMIT_CPU, 900),
+                     (resource.RLIMIT_FSIZE, 1024**3), (resource.RLIMIT_NOFILE, 1024),
+                     (resource.RLIMIT_NPROC, 1024), (resource.RLIMIT_CORE, 0)]:

+def archive(name, kind=None, data=b'public result'):
+    stream = io.BytesIO()
+    with tarfile.open(fileobj=stream, mode='w') as tar:
+        entry = tarfile.TarInfo(name)
+        if kind is not None:
+            entry.type = kind
+            entry.linkname = '/outside'
+        else:
+            entry.size = len(data)
+        tar.addfile(entry, io.BytesIO(data) if kind is None else None)
+    return stream.getvalue()
