from pathlib import Path
import json,hashlib,subprocess,re,html,collections,unittest,yaml
P=Path('tasks/052/evidence/assessment-20261115');O=P/'outputs';T=Path('target/quality-task-052')
def read(p):return json.loads(p.read_text())
def valid_scorecard(d,exit_code,stderr):
 if exit_code!=0 or stderr.strip() or not isinstance(d,dict):raise ValueError('invalid execution')
 names=[x['name'] for x in d.get('checks',[])];expected={'CI-Tests','Pinned-Dependencies','Signed-Releases','CII-Best-Practices','Security-Policy','Dangerous-Workflow','Fuzzing','SAST','Vulnerabilities','Token-Permissions','Code-Review','License','Maintained','Contributors','Dependency-Update-Tool','Branch-Protection','Packaging','Binary-Artifacts'}
 if set(names)!=expected or len(names)!=18:raise ValueError('missing or duplicate checks')
 if not re.fullmatch('[0-9a-f]{40}',d.get('repo',{}).get('commit','')):raise ValueError('missing target identity')
 return True
class AssessmentValidation(unittest.TestCase):
 def test_01_all_required_deliverables_and_mirrors(self):
  for name in ['best-practices.md','github-controls.json','process-review.md','scorecard.json','security-process.md']:
   self.assertTrue((O/name).is_file());self.assertEqual((O/name).read_bytes(),(T/name).read_bytes())
 def test_02_official_criterion_set_exactly_matches(self):
  expected={(l,k) for l,ss in yaml.safe_load((P/'official/badge-criteria.yml').read_text()) for s,gg in ss for g,cc in gg for k,m in cc};rows=read(O/'best-practices-assessment.json')['criteria']
  self.assertEqual(len(expected),145);self.assertEqual({(r['level'],r['criterion']) for r in rows},expected);self.assertEqual(len(rows),145)
  self.assertEqual(collections.Counter(r['level'] for r in rows),{'0':67,'1':55,'2':23})
 def test_03_missing_or_duplicate_criterion_rejected(self):
  rows=read(O/'best-practices-assessment.json')['criteria'];keys=[r['key'] for r in rows];self.assertNotEqual(set(keys[:-1]),set(keys));bad=keys+[keys[0]];self.assertNotEqual(len(bad),len(set(bad)))
 def test_04_official_live_and_pinned_descriptions_agree(self):
  def norm(s):return re.sub(r'\s+',' ',html.unescape(re.sub('<[^>]+>',' ',s))).strip()
  for r in read(P/'official/CRITERIA-EXTRACT.json'):self.assertIn(norm(r['pinnedDescription']),norm(r['text']))
 def test_05_all_criteria_have_owner_documents_execution_and_evidence(self):
  refs=read(O/'EVIDENCE-REFERENCES.json');rows=read(O/'best-practices-assessment.json')['criteria']
  for r in rows:
   for field in ['owner','documentation','executionEvidence','assessment','evidence']:self.assertTrue(r[field],(r['key'],field))
   for key in r['evidence']:
    self.assertIn(key,refs)
    for path in refs[key]:self.assertTrue(Path(path).is_file(),path)
 def test_06_not_applicable_has_official_permission(self):
  official={r['level']+'.'+r['id']:r for r in read(P/'official/CRITERIA-EXTRACT.json')}
  for r in read(O/'best-practices-assessment.json')['criteria']:
   if r['status']=='Not applicable':self.assertTrue(official[r['key']]['metadata'].get('na_allowed'),r['key']);self.assertTrue(r['assessment'])
 def test_07_scorecard_actual_run_and_tool_identity(self):
  d=read(O/'scorecard.json');r=read(P/'scorecard/FINAL-RUN.json');self.assertTrue(valid_scorecard(d,r['exitCode'],(P/'scorecard/final.stderr').read_text()));self.assertEqual(d['scorecard'],{'version':'v5.6.0','commit':'61e84743356275f6ffca23b1b2f2cb945e19eb96'});self.assertEqual(d['repo']['commit'],'f07d782cd68cfac95d93585d5ace049f590160d4');self.assertEqual(d['score'],4.3);self.assertEqual((O/'scorecard.json').read_bytes(),(P/'scorecard/final.stdout').read_bytes())
 def test_08_empty_exit_zero_and_bad_run_rejected(self):
  with self.assertRaises(ValueError):valid_scorecard({},0,'')
  with self.assertRaises(ValueError):valid_scorecard(read(P/'scorecard/authenticated.stdout'),1,(P/'scorecard/authenticated.stderr').read_text())
  bad=read(O/'scorecard.json');bad['checks'].pop()
