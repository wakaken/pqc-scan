"""Verify actual GitHub receipts/ZIP bytes and source; no stored commands run."""
import argparse
import datetime
import hashlib
import json
from pathlib import Path, PurePosixPath
import stat
import subprocess
import sys
import zipfile

parser=argparse.ArgumentParser()
parser.add_argument('run_dir',type=Path)
parser.add_argument('--commit',required=True)
parser.add_argument('--worktree',type=Path,required=True)
args=parser.parse_args()
root=Path.cwd();out=args.run_dir
load=lambda p:json.loads(Path(p).read_text())
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
sys.path.insert(0,str(root/'tests/quality'))
import evidence
run=load(out/'RUN-final.json');jobs=load(out/'JOBS-final.json')['jobs'];metadata=load(out/'ARTIFACTS-final.json')['artifacts']
assert run['head_sha']==args.commit and run['status']=='completed' and run['conclusion']=='success'
assert run['event']=='workflow_dispatch'
assert {j['name'] for j in jobs}=={'quality','baseline','scan','protocols'}
for job in jobs:
    if job['name']=='protocols':assert job['conclusion']=='skipped'
    else:
        assert job['status']=='completed' and job['conclusion']=='success'
        assert job['steps'] and all(s['conclusion']=='success' for s in job['steps'])
remote_commit=load(out/'REMOTE-COMMIT.json')
local_tree=subprocess.check_output(['git','-C',str(args.worktree),'rev-parse',args.commit+'^{tree}'],text=True).strip()
assert remote_commit['sha']==args.commit and remote_commit['commit']['tree']['sha']==local_tree
catalog_bytes=subprocess.check_output(['git','-C',str(args.worktree),'show',args.commit+':docs/quality/EVIDENCE-CATALOG.json'])
catalog=json.loads(catalog_bytes)
current=load('docs/quality/EVIDENCE-CATALOG.json')
assert catalog['requirements']==current['requirements'] and catalog['version']==current['version']
assert [r for r in catalog['records'] if r['task']!='053']==[r for r in current['records'] if r['task']!='053']
own=catalog['records'][-1];assert own['task']=='053'
current_own=next(r for r in current['records'] if r['task']=='053')
assert current_own['sourceBindings']==own['sourceBindings']
assert current_own['measurement']['populationDigest']==own['measurement']['populationDigest']
input_snapshot=out/'RUN-INPUT-CATALOG.json'
if input_snapshot.exists():assert input_snapshot.read_bytes()==catalog_bytes
else:input_snapshot.write_bytes(catalog_bytes)
for p,h in own['sourceBindings'].items():
    assert sha(p)==h,p
    committed=subprocess.check_output(['git','-C',str(args.worktree),'show',args.commit+':'+p])
    assert hashlib.sha256(committed).hexdigest()==h,p
assert {a['name'] for a in metadata}=={'pqc-quality-integrated-pr','pqc-quality-baseline','pqc-self-scan-evidence','pqc-dependency-sbom'}
artifacts=[];integrated=[];inventories={}
for item in metadata:
    path=out/(item['name']+'.zip')
    assert path.stat().st_size==item['size_in_bytes']
    assert sha(path)==item['digest'].removeprefix('sha256:')
    assert item['workflow_run']['head_sha']==args.commit and item['workflow_run']['id']==run['id'] and not item['expired']
    is_integration=item['name'] in {'pqc-quality-integrated-pr','pqc-quality-baseline'}
    if is_integration:
        days=(datetime.datetime.fromisoformat(item['expires_at'].replace('Z','+00:00'))-datetime.datetime.fromisoformat(item['created_at'].replace('Z','+00:00'))).total_seconds()/86400
        assert 29.9<days<=30.1
    with zipfile.ZipFile(path) as archive:
        infos=archive.infolist();assert len(infos)<10000 and sum(i.file_size for i in infos)<1024**3
        assert len({i.filename for i in infos})==len(infos)
        for info in infos:
            p=PurePosixPath(info.filename)
            assert not p.is_absolute() and '..' not in p.parts and '\\' not in info.filename
            assert ':' not in info.filename and not any(ord(c)<32 for c in info.filename)
            assert stat.S_IFMT(info.external_attr>>16) in {0,stat.S_IFREG,stat.S_IFDIR}
        inventories[item['name']]=[{'path':i.filename,'size':i.file_size,'crc32':i.CRC} for i in infos]
        if is_integration:
            prefix='' if item['name']=='pqc-quality-integrated-pr' else 'quality-integrated/'
            files={};selected=0
            for name in ['quality-evidence-index.json','test-evidence-index.json','quality-assessment.md','ci-measurement-plan.md']:
                info=archive.getinfo(prefix+name);selected+=info.file_size
                assert info.file_size<=64*1024**2 and selected<=32*1024**2
                files[name]=archive.read(prefix+name)
                dest=out/item['name']/name;dest.parent.mkdir(exist_ok=True);dest.write_bytes(files[name])
            integrated.append(files)
    artifacts.append({'id':item['id'],'name':item['name'],'zipBytes':path.stat().st_size,'sha256':sha(path),'githubDigestMatches':True,'candidateCommitMatches':True,'members':len(infos),'regularSafePaths':True})
assert len(integrated)==2 and integrated[0]==integrated[1]
remote=json.loads(integrated[0]['quality-evidence-index.json'])
expected=evidence.build_index(root/input_snapshot,root);expected['target']['head']=args.commit
expected['catalog']['path']='docs/quality/EVIDENCE-CATALOG.json'
assert expected==remote
assert remote['catalog']['sha256']==hashlib.sha256(catalog_bytes).hexdigest()
assert len(remote['requirements'])==73 and len(remote['records'])==19
assert json.loads(integrated[0]['test-evidence-index.json'])['records']==remote['testEvidence']
text=integrated[0]['quality-assessment.md'].decode()
for row in remote['requirements']:assert f"| {row['id']} | {row['status']} | {row['ownerTask']} |" in text
assert integrated[0]['ci-measurement-plan.md']==Path('docs/quality/CI-MEASUREMENT-PLAN.md').read_bytes()
receipt={'status':'Verified','runId':run['id'],'runUrl':run['html_url'],'runAttempt':run['run_attempt'],'commit':args.commit,'tree':local_tree,'event':run['event'],'runConclusion':run['conclusion'],
         'sourceBindings':own['sourceBindings'],'populationDigest':own['measurement']['populationDigest'],
         'enabledJobs':{j['name']:j['conclusion'] for j in jobs if j['name']!='protocols'},'protocols':'skipped','requiredStepsVerified':True,'artifacts':artifacts,
         'artifactDigestsVerified':True,'qualityBaselineFourOutputsByteEqual':True,'localRemoteFactsEqualExceptActualCommit':True,'retentionDays':30,
         'catalogSha256':remote['catalog']['sha256'],'inputCatalogSnapshot':str(input_snapshot),'verifierSourceSha256':sha(Path(__file__)),'requirements':73,'records':19,
         'readPolicy':'RawZIP digest and bounded safe inventory; only4 integrated files/job decompressed under64MiB/file and32MiB aggregate. Large original benchmark bodies are not extracted.'}
(out/'HOSTED-VERIFICATION.json').write_text(json.dumps(receipt,indent=2)+'\n')
(out/'ZIP-INVENTORIES.json').write_text(json.dumps(inventories,indent=2)+'\n')
print('Verified exact-source hosted run',run['id'],'all3jobs/steps,4 ZIPs,source/population,73rows and allreport facts.')
