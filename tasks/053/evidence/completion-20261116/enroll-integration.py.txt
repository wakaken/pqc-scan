"""Enroll actual detached integration evidence; never invent hosted execution."""
import argparse
import hashlib
import json
from pathlib import Path
import sys

root=Path.cwd()
sys.path.insert(0,str(root/'tests/quality'))
import evidence

parser=argparse.ArgumentParser()
parser.add_argument('--checks',type=Path)
parser.add_argument('--implementation-review',type=Path)
parser.add_argument('--test-review',type=Path)
args=parser.parse_args()
base=Path('tasks/053/evidence/completion-20261116')
catalog_path=Path('docs/quality/EVIDENCE-CATALOG.json')
# The immutable pre-correction catalog contains all18 upstream studies.
upstream=json.loads((base/'before/docs/quality/EVIDENCE-CATALOG.json').read_text())
assert len(upstream['records'])==18
population=evidence.binding_digest(upstream)
files=['.github/workflows/pqc-scan.yml','tests/quality/report.py',
       'tests/quality/evidence.py','tests/quality/test_evidence.py','tests/quality/test_ci053.py',
       'tests/quality/README.md','docs/quality/CI-MEASUREMENT-PLAN.md',
       'README.md','README.ja.md',str(base/'enroll-integration.py.txt'),
       str(base/'before/docs/quality/EVIDENCE-CATALOG.json')]
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
bindings={p:sha(p) for p in files}
selected=set(files)
commands=[dict(argv=['python3','-m','unittest','discover','-s','tests/quality','-p','test_*.py'],
               exitCode=None,reason='Current final-source validation is still pending')]
execution='Not Tested';criteria={q:'Partial' for q in ['Q6-01','Q6-02','Q6-16','Q6-18']}
if args.checks:
    checks=json.loads(args.checks.read_text())
    assert checks['sourceBindings']==bindings,'Validation receipt binds different source bytes'
    assert checks['populationDigest']==population
    selected.add(str(args.checks));commands=checks['commands'];execution=checks['executionStatus']
    if execution=='Passed':
        criteria.update({'Q6-02':'Passed','Q6-16':'Passed','Q6-18':'Passed'})
reviews=[]
for stage,path in [('implementation',args.implementation_review),('test',args.test_review)]:
    if path:
        review=json.loads(path.read_text())
        assert review['stage']==stage and review['status']=='accepted'
        assert review['sourceBindings']==bindings and review['populationDigest']==population
        assert review['sourceDigest']==evidence.binding_digest(bindings)
        selected.add(str(path))
        reviews.append(dict(review,artifact=str(path)))
artifacts=[dict(path=p,sha256=sha(p),purpose='Actual integration source, immutable input population or detached validation/review receipt') for p in sorted(selected)]
observations=[]
if args.checks:
    for label in ['python','node','localCI','hostedCI']:
        observations.append(dict(label=label,artifact=str(args.checks),pointer='/'+label))
record=dict(id='task053-integration-validation',task='053',requirements=list(criteria),
  result=dict(status='Partial',criteria=criteria,
              summary='Integration validation and independent review evidence; final-source hosted jobs/artifacts remain unverified',
              originalOutcome='Local validation and hosted CI are assessed separately; no release acceptance'),
  measurement=dict(head=__import__('subprocess').check_output(['git','rev-parse','HEAD'],text=True).strip(),
       version='0.2.0 source / v0.3 quality integration',features='Offline all-axis catalog and baseline-only/combined report modes',
       standards='Original Q6-01/Q6-02/Q6-16/Q6-18; no new score or release gate',
       method='Actual unit/negative controls, local changed CI steps and independent source-bound review; no stored command execution',
       environment='macOS arm64 local validation; actual hosted environment remains Not Tested',date='2026-11-16',
       comparison='Corrects original053 own-evidence omission; previous observations remain historical',
       executionStatus=execution,commands=commands,populationDigest=population),
  sourceBindings=bindings,artifacts=artifacts,observations=observations,reviews=reviews,
  limitations=['Q6-01 stays Partial until final-source hosted job/artifact evidence is verified.',
               'Record exclusion avoids a circular input self-hash; the complete emitted catalog retains its full SHA-256.',
               'This record does not change unfinished052 or start054; no overall product/release pass is asserted.'],
  concerns=[dict(priority='high',text='Final-source hosted CI and uploaded artifact bytes have not been verified.')])
catalog=dict(upstream,records=upstream['records']+[record])
catalog_path.write_text(json.dumps(catalog,indent=2,ensure_ascii=False)+'\n')
print('Enrolled053:',len(catalog['records']),'records; hosted criterion remains Partial; review stages',len(reviews))
