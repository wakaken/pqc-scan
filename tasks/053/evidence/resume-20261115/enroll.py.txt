def hash_map(data):
    if not isinstance(data,dict):
        return {}
    for key in ['source_inputs','source','files','protectedBindings','bindings','sourceBindings']:
        value=data.get(key)
        if isinstance(value,dict):
            found={p:h for p,h in value.items() if isinstance(p,str) and isinstance(h,str) and re.fullmatch('[0-9a-f]{64}',h)}
            if found:
                return found
    if data and all(isinstance(h,str) and re.fullmatch('[0-9a-f]{64}',h) for h in data.values()):
        return data
    return {}

def commands(value):
    found=[]
    if isinstance(value,dict):
        argv=value.get('argv', value.get('command'))
        code=value.get('exitCode',value.get('exit',value.get('returncode')))
        if isinstance(argv,str) and type(code) is int:
            found.append({'argv':['sh','-c',argv], 'exitCode':code})
        elif isinstance(argv,list) and argv and all(isinstance(x,str) for x in argv) and type(code) is int:
            found.append({'argv':argv,'exitCode':code})
        for item in value.values():found.extend(commands(item))
    elif isinstance(value,list):
        for item in value:found.extend(commands(item))
    return found
